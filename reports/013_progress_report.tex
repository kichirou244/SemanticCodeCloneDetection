\section{Các phương pháp}

\begin{itemize}
\item \textbf{MOSS} (baseline). Schleimer, Wilkerson \& Aiken 2003. So khớp đặc trưng
      k-gram. Tham số k=20, w=5 (EXP-013). Sau đó (báo cáo [06-10]) baseline chính
      chuyển sang MOSS service thật (EXP-014).
\item \textbf{CodeBERT}. \texttt{microsoft/\allowbreak codebert-base}. Ghép ba
      vector (A, B, độ lệch tuyệt đối A với B) rồi qua một lớp MLP.
\item \textbf{GraphCodeBERT}. \texttt{microsoft/\allowbreak graphcodebert-base}.
      Như CodeBERT, thêm quan hệ token qua cây cú pháp và dữ liệu luồng. Giữ nguyên
      siêu tham số CodeBERT.
\item \textbf{ASTNN}. Câu cú pháp rút gọn thành đồ thị câu lệnh, qua mạng đệ quy
      và mạng hai chiều. Huấn luyện từ đầu.
\item \textbf{CPG-GNN}. Đồ thị thuộc tính dạng code property graph (6 loại cạnh),
      qua mạng GAT 3 tầng rồi đầu MLP như ba phương pháp trên. Huấn luyện từ đầu.
\end{itemize}

\section{Hạn chế}

\begin{itemize}
\item Mỗi phương pháp chạy một lần với seed 42.
\item ASTNN dừng ở epoch 7 (early stopping, tốt nhất epoch 4).
\item CPG-GNN dừng ở epoch 5 (hết epoch, val F1 còn tăng).
\item CodeBERT, GraphCodeBERT cắt hàm còn 256 token; 78,6\% hàm trong tập train bị
      cắt.
\end{itemize}

\section{Kế hoạch tiếp theo}

\item Thử độ dài đầu vào 512 token.
